1 Setup

rm(list = ls())
source("~/repositories/cwb-cads/R-utils/api.R")
url_api <- "https://corpora.linguistik.uni-erlangen.de/cwb-cads-spheroscope/"
  • get API access token via /user/login
cads_connect(user = "admin", password = "RC21", url = url_api)
cads_list_corpora()

slots <- list(
  list(slot = "verb", start = "0", end = "0"),
  list(slot = "rest", start = "1", end = "matchend")
)
slot.query.1 <- cads_mk_request("spheroscope/slot-query/create") |>
  req_body_json(list(
      cqp_query = '("ich"%c|"er"%c) @0[lemma_smor="wollen"] @1[xpos="VVINF"]',
      name = 'ich oder er',
      corpus_id = 1,  # GERMAPARL220WEB
      slots = slots,
      corrections = list()
  )) |>
  cads_perform_request()
